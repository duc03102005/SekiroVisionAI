cmake_minimum_required(VERSION 3.25)
project(SekiroVisionAI VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CTest)
if(BUILD_TESTING)
  add_executable(CaptureTimingTests Tests/CaptureTimingTests.cpp)
  target_include_directories(CaptureTimingTests PRIVATE CaptureEngine/include)
  add_test(NAME capture_timing COMMAND CaptureTimingTests)
endif()

if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "The Windows capture target requires MSVC and the Windows SDK.")
  endif()
  enable_language(RC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_library(CaptureEngine STATIC CaptureEngine/src/CaptureEngine.cpp)
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE SVAI_BUILD_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(NOT SVAI_BUILD_REVISION)
    set(SVAI_BUILD_REVISION "unknown")
  endif()
  target_include_directories(CaptureEngine PUBLIC CaptureEngine/include)
  target_compile_definitions(CaptureEngine PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_definitions(CaptureEngine PRIVATE _WIN32_WINNT=0x0A00)
  target_compile_definitions(CaptureEngine PRIVATE SVAI_BUILD_REVISION="${SVAI_BUILD_REVISION}")
  target_compile_options(CaptureEngine PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_libraries(CaptureEngine PUBLIC d3d11 dxgi windowsapp)

  add_executable(SekiroCaptureProbe WIN32 App/CaptureProbe.cpp App/CaptureProbe.rc)
  target_compile_options(SekiroCaptureProbe PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_options(SekiroCaptureProbe PRIVATE /MANIFEST:NO)
  target_link_libraries(SekiroCaptureProbe PRIVATE CaptureEngine comdlg32 psapi)
else()
  message(STATUS "Windows capture is unavailable here; only portable timing tests are built.")
endif()
