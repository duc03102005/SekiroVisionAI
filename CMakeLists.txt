cmake_minimum_required(VERSION 3.25)
project(SekiroVisionAI VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CTest)
add_library(MvpAlgorithms INTERFACE)
target_include_directories(MvpAlgorithms INTERFACE CaptureEngine/include VisionEngine/include ThreatEngine/include ModelRuntime/include)
if(BUILD_TESTING)
  add_executable(CaptureTimingTests Tests/CaptureTimingTests.cpp)
  target_include_directories(CaptureTimingTests PRIVATE CaptureEngine/include)
  add_test(NAME capture_timing COMMAND CaptureTimingTests)
  add_executable(AutoDodgeTests Tests/AutoDodgeTests.cpp)
  target_link_libraries(AutoDodgeTests PRIVATE MvpAlgorithms)
  add_test(NAME auto_dodge_behavior COMMAND AutoDodgeTests)
  add_executable(TemporalDecisionTests Tests/TemporalDecisionTests.cpp)
  target_link_libraries(TemporalDecisionTests PRIVATE MvpAlgorithms)
  add_test(NAME temporal_decision_behavior COMMAND TemporalDecisionTests)
endif()

if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "The Windows capture target requires MSVC and the Windows SDK.")
  endif()
  enable_language(RC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_library(CaptureEngine STATIC CaptureEngine/src/CaptureEngine.cpp)
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE SVAI_BUILD_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(NOT SVAI_BUILD_REVISION)
    set(SVAI_BUILD_REVISION "unknown")
  endif()
  target_include_directories(CaptureEngine PUBLIC CaptureEngine/include)
  target_compile_definitions(CaptureEngine PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_definitions(CaptureEngine PRIVATE _WIN32_WINNT=0x0A00)
  target_compile_definitions(CaptureEngine PRIVATE SVAI_BUILD_REVISION="${SVAI_BUILD_REVISION}")
  target_compile_options(CaptureEngine PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_libraries(CaptureEngine PUBLIC d3d11 d3dcompiler dxgi windowsapp)

  include(cmake/OnnxRuntime.cmake)
  add_library(ModelRuntime STATIC ModelRuntime/src/TemporalModel.cpp)
  target_link_libraries(ModelRuntime PUBLIC MvpAlgorithms PRIVATE OnnxRuntime)
  target_compile_options(ModelRuntime PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_compile_definitions(ModelRuntime PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(SVAI_ORT_FLAVOR STREQUAL "DIRECTML")
    target_compile_definitions(ModelRuntime PRIVATE SVAI_ORT_DML=1)
    target_include_directories(ModelRuntime SYSTEM PRIVATE "${svai_directml_SOURCE_DIR}/include")
  endif()
  add_library(SampleRecording STATIC RecordingEngine/src/SampleRecorder.cpp)
  target_include_directories(SampleRecording PUBLIC RecordingEngine/include)
  target_link_libraries(SampleRecording PUBLIC CaptureEngine windowscodecs ole32)
  target_compile_options(SampleRecording PRIVATE /W4 /permissive- /EHsc /utf-8)

  add_library(AutoDodgeEngine STATIC DodgeEngine/src/AutoDodge.cpp InputEngine/src/InputService.cpp)
  target_include_directories(AutoDodgeEngine PUBLIC DodgeEngine/include InputEngine/include)
  target_link_libraries(AutoDodgeEngine PUBLIC CaptureEngine MvpAlgorithms ModelRuntime SampleRecording shell32 ole32 user32 comdlg32)
  target_compile_options(AutoDodgeEngine PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_compile_definitions(AutoDodgeEngine PRIVATE SVAI_BUILD_REVISION="${SVAI_BUILD_REVISION}")

  add_executable(SekiroAutoDodge WIN32 App/AutoDodgeApp.cpp App/AutoDodge.rc)
  target_link_libraries(SekiroAutoDodge PRIVATE AutoDodgeEngine)
  target_compile_options(SekiroAutoDodge PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_options(SekiroAutoDodge PRIVATE /MANIFEST:NO)
  svai_copy_runtime(SekiroAutoDodge)
  if(BUILD_TESTING)
    add_executable(GpuReadbackTests Tests/GpuReadbackTests.cpp)
    target_link_libraries(GpuReadbackTests PRIVATE CaptureEngine)
    add_test(NAME gpu_readback_functional COMMAND GpuReadbackTests)
    add_executable(ModelRuntimeTests Tests/ModelRuntimeTests.cpp)
    target_link_libraries(ModelRuntimeTests PRIVATE ModelRuntime)
    svai_copy_runtime(ModelRuntimeTests)
    add_test(NAME model_runtime_functional COMMAND ModelRuntimeTests "${CMAKE_SOURCE_DIR}/out/testdata/temporal-fixture.onnx")
    add_executable(SampleRecorderTests Tests/SampleRecorderTests.cpp)
    target_link_libraries(SampleRecorderTests PRIVATE SampleRecording)
    add_test(NAME sample_recording_functional COMMAND SampleRecorderTests)
  endif()

  add_executable(SekiroCaptureProbe WIN32 App/CaptureProbe.cpp App/CaptureProbe.rc)
  target_compile_options(SekiroCaptureProbe PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_options(SekiroCaptureProbe PRIVATE /MANIFEST:NO)
  target_link_libraries(SekiroCaptureProbe PRIVATE CaptureEngine comdlg32 psapi)
else()
  message(STATUS "Windows capture is unavailable here; only portable timing tests are built.")
endif()
